cmake_minimum_required(VERSION 3.22)
project(PhantomTune VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(PhantomTune COMPANY_NAME "Phantom Culture" BUNDLE_ID "com.phantomculture.phantomtune" PLUGIN_MANUFACTURER_CODE PhCu PLUGIN_CODE PhTu FORMATS VST3 PRODUCT_NAME "Phantom Tune" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(PhantomTune PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(PhantomTune PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(PhantomTune PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
